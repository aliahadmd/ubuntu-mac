# Kernel and matching headers (v4l2loopback DKMS builds against them).
# The leading ! installs without recommends so the kernel never pulls
# grub-efi-arm64 or flash-kernel: the factory image has no partition table
# and is direct-booted, so no bootloader may ship. (Device firmware still
# arrives as a hard Depends of linux-modules and is accepted.)
!linux-image-generic
!linux-headers-generic

initramfs-tools

openssh-server

whiptail

ubuntu-desktop-minimal
